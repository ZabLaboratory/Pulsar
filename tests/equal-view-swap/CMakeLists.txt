cmake_minimum_required(VERSION 3.28)
project(equal_view_swap LANGUAGES C)
enable_testing()
if(NOT DEFINED OBS_SOURCE_DIR)
  message(FATAL_ERROR "OBS_SOURCE_DIR must point to the patched libobs source tree")
endif()
file(READ "${OBS_SOURCE_DIR}/libobs/obs-view.c" VIEW_SOURCE)
string(FIND "${VIEW_SOURCE}" "void obs_view_apply_pending_atomic_swap(" FUNCTION_START)
string(FIND "${VIEW_SOURCE}" "void obs_view_render(" FUNCTION_END)
math(EXPR FUNCTION_SIZE "${FUNCTION_END} - ${FUNCTION_START}")
string(SUBSTRING "${VIEW_SOURCE}" ${FUNCTION_START} ${FUNCTION_SIZE} SWAP_IMPLEMENTATION)
configure_file(swap-test.c.in swap-test.c @ONLY)
add_executable(equal-view-swap "${CMAKE_CURRENT_BINARY_DIR}/swap-test.c")
add_test(NAME equal-view-swap COMMAND equal-view-swap)
